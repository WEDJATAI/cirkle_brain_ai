    59→  const abortRef = React.useRef<AbortController | null>(null);

    61→
    62→  React.useEffect(() => {
    63→    apiGet<{ platforms: Array<{ slug: string; displayName: string; domain: string; status: string }> }>("/api/brain/platforms")
    64→      .then((d) => setPlatforms(d.platforms.filter((p) => p.status === "ACTIVE")))
    65→      .catch(() => {});
    66→  }, []);
    67→
    68→  React.useEffect(() => {
    69→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" });

    71→
    72→  async function send(text?: string) {
    73→    const content = (text ?? input).trim();
    74→    if (!content || streaming) return;
    75→    setInput("");
    76→    const userMsg: ChatMessage = { id: crypto.randomUUID(), role: "user", content, createdAt: Date.now() };
    77→    const assistantMsg: ChatMessage = { id: crypto.randomUUID(), role: "assistant", content: "", state: initialStreamState(), question: content, createdAt: Date.now() };
    78→    setMessages((m) => [...m, userMsg, assistantMsg]);
    79→    setActiveMessageId(assistantMsg.id);
    80→    setStreaming(true);
    81→    abortRef.current = new AbortController();
    82→    try {
    83→      await streamBrainResponse(
    84→        { input: { text: content }, mode, platformSlug },
    85→        (state) => {
    86→          setMessages((prev) => prev.map((m) => (m.id === assistantMsg.id ? { ...m, content: state.tokens || (state.error ? `⚠️ ${state.error.message}` : ""), state: { ...state } } : m)));
    87→        },
    88→        abortRef.current.signal,
    89→      );
    90→    } catch (err: any) {
    91→      setMessages((prev) => prev.map((m) => (m.id === assistantMsg.id ? { ...m, content: `⚠️ Stream failed: ${err?.message ?? err}` } : m)));
    92→    } finally {
    93→      setStreaming(false);
    94→      // Keep activeMessageId pointing at the last assistant message so the
    95→      // trace / evidence / tools / memory panels stay populated.
    96→    }

   291→  if (message.role === "user") {
   292→    return (
   293→      <div className="flex justify-end animate-fade-up">

   296→          <div
   297→            className="pointer-events-none absolute inset-0 opacity-20"
   298→            style={{ backgroundImage: "var(--gradient-hero)" }}
   299→            aria-hidden
   300→          />
   301→          <span className="relative">{message.content}</span>
   302→        </div>
   303→      </div>
   304→    );
   305→  }
   306→  const state = message.state;
   307→  const streaming = active && !state?.done;
   308→  return (
   309→    <div className="flex flex-col gap-2 animate-fade-up">
   310→      <div className="flex items-start gap-2">
   311→        <div className="orbit-ring mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-lg !rounded-xl">
   312→          <CirkleMark size={20} aria-hidden className="h-5 w-5" />

   316→            <div className="orbit-ring relative overflow-hidden rounded-[22px] rounded-tl-md px-3 py-2 text-sm leading-relaxed">

   318→              {streaming && <span className="ml-0.5 inline-block h-3.5 w-1 cirkle-pulse bg-[color:var(--color-cirkle-cyan)] align-middle" />}

   355→function FeedbackButtons({ requestId, answer, question }: { requestId: string; answer: string; question?: string }) {
   356→  const [given, setGiven] = React.useState<"thumbs_up" | "thumbs_down" | null>(null);
   357→  const [loading, setLoading] = React.useState(false);
   358→
   359→  async function sendFeedback(signal: "thumbs_up" | "thumbs_down") {
   360→    if (given || loading) return;
   361→    setLoading(true);
   362→    setGiven(signal);
   363→    try {
   364→      await apiPost("/api/brain/feedback", {
   365→        requestId, signal, answer: answer.slice(0, 500), question: question?.slice(0, 500),
   366→      });
   367→    } catch {
   368→      // revert on failure
   369→      setGiven(null);
   370→    } finally {
   371→      setLoading(false);
   372→    }
   373→  }
   374→
   375→  return (
   376→    <div className="flex items-center gap-1">
   377→      <button
   378→        onClick={() => sendFeedback("thumbs_up")}
   379→        disabled={!!given || loading}
   380→        className={cn(
   381→          "flex h-6 w-6 items-center justify-center rounded-md border text-[10px] transition-colors",
   382→          given === "thumbs_up"
   383→            ? "border-emerald-500/40 bg-emerald-500/15 text-emerald-600"
   384→            : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   385→        )}
   386→        title="Good answer"
   387→      >
   388→        <ThumbsUp className="h-3 w-3" />
   389→      </button>
   390→      <button
   391→        onClick={() => sendFeedback("thumbs_down")}
   392→        disabled={!!given || loading}
   393→        className={cn(
   394→          "flex h-6 w-6 items-center justify-center rounded-md border text-[10px] transition-colors",
   395→          given === "thumbs_down"
   396→            ? "border-rose-500/40 bg-rose-500/15 text-rose-600"
   397→            : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   398→        )}
   399→        title="Bad answer"
   400→      >
   401→        <ThumbsDown className="h-3 w-3" />
   402→      </button>
   403→      {given && (
   404→        <span className="text-[10px] text-muted-foreground">
   405→          {given === "thumbs_up" ? "Thanks — helps the Brain learn" : "Thanks — Brain will investigate"}
   406→        </span>
   407→      )}
   408→    </div>
   409→  );
   410→}

   412→function ResponseChips({ state }: { state: BrainStreamState }) {
   413→  const chips: React.ReactNode[] = [];
   414→  if (state.model) {
   415→    chips.push(
   416→      <Chip key="model" icon={<Cpu className="h-3 w-3" />} label={`${state.model.model}${state.model.fallbackUsed ? " (fallback)" : ""}`} tone={state.model.fallbackUsed ? "warn" : "ok"} />
   417→    );
   418→  }
   419→  if (state.research) {
   420→    chips.push(<Chip key="research" icon={<Globe className="h-3 w-3" />} label={`web: ${state.research.ingestedCount} learned`} tone="info" />);
   421→  }
   422→  if (state.verification) {
   423→    chips.push(<Chip key="ver" icon={<ShieldCheck className="h-3 w-3" />} label={state.verification.status} tone={verificationTone(state.verification.status)} />);
   424→  }
   425→  if (state.tools.length > 0) {
   426→    chips.push(<Chip key="tools" icon={<Wrench className="h-3 w-3" />} label={`${state.tools.length} tool`} tone="ok" />);
   427→  }
   428→  if (state.evidence.length > 0) {
   429→    chips.push(<Chip key="ev" icon={<FileText className="h-3 w-3" />} label={`${state.evidence.length} evidence`} tone="ok" />);
   430→  }
   431→  if (state.cost) {
   432→    chips.push(<Chip key="cost" icon={<Zap className="h-3 w-3" />} label={`${state.cost.latencyMs}ms`} tone="muted" />);
   433→    if (state.cost.costUsd > 0) chips.push(<Chip key="cost2" icon={<DollarSign className="h-3 w-3" />} label={`$${state.cost.costUsd.toFixed(4)}`} tone="muted" />);
   434→  }
   435→  if (chips.length === 0) return null;
   436→  return <div className="flex flex-wrap items-center gap-1">{chips}</div>;
   437→}

   439→function Chip({ icon, label, tone }: { icon: React.ReactNode; label: string; tone: "ok" | "warn" | "err" | "muted" | "info" }) {

   599→function ToolCard({ tool }: { tool: ToolResult }) {
   600→  const [approving, setApproving] = React.useState(false);
   601→  const [current, setCurrent] = React.useState(tool);
   602→  React.useEffect(() => { setCurrent(tool); }, [tool]);
   603→  async function approve() {
   604→    setApproving(true);
   605→    try {
   606→      const r = await apiPost<{ result: ToolResult }>("/api/brain/tools/approve", { executionId: current.executionId, approver: "alice@acme.test" });
   607→      setCurrent(r.result);
   608→      toast.success(`Approved ${current.toolId} → ${r.result.state}`);
   609→    } catch (e: any) {
   610→      toast.error(e?.message ?? "approval failed");
   611→    } finally {
   612→      setApproving(false);
   613→    }
   614→  }
   615→  return (
   616→    <div className="rounded-lg border bg-card p-2.5">
   617→      <div className="mb-1 flex items-center justify-between gap-2">
   618→        <div className="flex items-center gap-1.5">
   619→          <Wrench className="h-3 w-3 text-[color:var(--color-cirkle-cyan)]" />
   620→          <span className="text-xs font-medium">{current.toolId}</span>
   621→        </div>
   622→        <Badge variant="outline" className={cn("text-[9px]", actionStateClass(current.state))}>{current.state}</Badge>
   623→      </div>
   624→      {current.output && (
   625→        <pre className="mt-1 max-h-32 overflow-auto rounded bg-muted/40 p-2 text-[10px] font-mono leading-tight">{JSON.stringify(current.output, null, 2)}</pre>
   626→      )}
   627→      {current.error && <p className="mt-1 text-[10px] text-rose-600">{current.error}</p>}
   628→      {current.requiresApproval && !current.approved && current.state === "AUTHORIZED" && (
   629→        <Button size="sm" className="mt-2 h-7 w-full text-xs" onClick={approve} disabled={approving}>
   630→          {approving ? <Loader2 className="mr-1 h-3 w-3 animate-spin" /> : <ShieldCheck className="mr-1 h-3 w-3" />}
   631→          Approve (human authorization)
   632→        </Button>
   633→      )}
   634→      <div className="mt-1 flex items-center justify-between text-[10px] text-muted-foreground">
   635→        <span>{current.durationMs}ms</span>
   636→        <span>governed execution</span>
   637→      </div>
   638→    </div>
   639→  );
   640→}
