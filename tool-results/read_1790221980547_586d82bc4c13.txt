   359→        slug: p.slug, name: p.name, displayName: p.displayName, domain: p.domain, description: p.description,
   360→        repoUrl: p.repoUrl, productionUrl: p.productionUrl,
   361→        knowledgeScope: p.knowledgeScope, memoryScope: p.memoryScope, toolScope: p.toolScope,
   362→        dataClassCeiling: p.dataClassCeiling, modelPolicy: p.modelPolicy,
   363→        allowedBrainScopes: p.allowedBrainScopes.join(","),
   364→        riskCeiling: p.riskCeiling,
   365→        capabilities: JSON.stringify(p.capabilities),
   366→        eventTypes: JSON.stringify(p.eventTypes),
   367→        domainTools: JSON.stringify(p.domainTools),
   368→        adapterStatus: p.slug === "mashahd" || p.slug === "verify" || p.slug === "judge_smart" ? "AUDITED" : "REGISTERED",
   369→        status: p.slug === "egycourt" ? "REGISTERED" : "ACTIVE",
   370→        personality: p.personality ? JSON.stringify(p.personality) : null,
   371→      },
   372→    });
   373→    // Register an adapter registration record (§27, §92) for each platform.
   374→    const platformRow = await db.platform.findUnique({ where: { slug: p.slug } });
   375→    if (platformRow) {
   376→      const existingAdapter = await db.adapterRegistration.findFirst({ where: { platformId: platformRow.id } });
   377→      if (!existingAdapter) {
   378→        await db.adapterRegistration.create({
   379→          data: {
   380→            platformId: platformRow.id,
   381→            adapterKind: "sdk:typescript",
   382→            sdkVersion: "0.1.0",
   383→            authMethod: "service_token",
   384→            serviceIdentity: `platform:${p.slug}:adapter`,
   385→            scopes: p.allowedBrainScopes.join(","),
   386→            lastHandshakeAt: new Date(),
   387→            status: "ACTIVE",
   388→          },
   389→        });
   390→      }
   391→    }
   392→    // Link the platform to the Acme/Mashahd application (demo mapping §16, §29).
   393→    const linkedApp = mashahd.id;
   394→    if (platformRow) {
   395→      const existingLink = await db.platformApplication.findUnique({ where: { platformId_applicationId: { platformId: platformRow.id, applicationId: linkedApp } } });
   396→      if (!existingLink) {
   397→        await db.platformApplication.create({ data: { platformId: platformRow.id, applicationId: linkedApp, tenantId: acme.id } });
   398→      }
   399→    }
   400→  }
   401→  created.platforms = PLATFORM_CATALOG.length;
   402→
   403→  // ----- Per-platform evaluation suites (§149) -----
   404→  for (const p of PLATFORM_CATALOG) {
   405→    const platformRow = await db.platform.findUnique({ where: { slug: p.slug } });
   406→    if (!platformRow) continue;
   407→    const existingSet = await db.platformEvaluationSet.findFirst({ where: { platformId: platformRow.id } });
   408→    if (existingSet) continue;
   409→    const set = await db.platformEvaluationSet.create({
   410→      data: { platformId: platformRow.id, name: `${p.name} Golden Set`, description: `Domain golden dataset for ${p.displayName} (§149)`, status: "ACTIVE" },
   411→    });
   412→    // Seed 2 representative cases per platform derived from its capabilities.
   413→    const cases = p.capabilities.slice(0, 2).map((cap) => ({
   414→      input: `(${p.slug}/${cap}) — sample domain task for ${p.displayName}`,
   415→      expected: cap,
   416→      tags: `${p.domain},${cap}`,
   417→    }));
   418→    for (const c of cases) {
   419→      await db.platformEvaluationCase.create({ data: { setId: set.id, input: c.input, expected: c.expected, tags: c.tags } });
   420→    }
   421→  }
   422→
   423→  // ----- Evaluation set (§86 golden dataset) -----
   424→  const evalSet = await db.evaluationSet.upsert({
   425→    where: { id: "golden-baseline" },
   426→    update: {},
   427→    create: { id: "golden-baseline", name: "Golden Baseline v1", description: "Phase 0 baseline (§78)", status: "ACTIVE" },
   428→  });
   429→  await db.evaluationCase.deleteMany({ where: { setId: evalSet.id } }).catch(() => {});
   430→  const cases = [
   431→    { input: "What is the capital of France?", expected: "Paris", tags: "simple,factual" },
   432→    { input: "What does the Wedjat Brain spec say about authorization?", expected: "Authorization is outside the model", tags: "knowledge,retrieval" },
   433→    { input: "What is invoice 1827?", expected: "$4,820", tags: "structured,invoice" },
   434→    { input: "Add 23 and 19", expected: "42", tags: "tool,calc" },
   435→    { input: "What is the weather in Dubai?", expected: "stubbed", tags: "tool,weather" },
   436→    { input: "Summarize why the model is not the brain", expected: "Brain owns the system around the model", tags: "reasoning,synthesis" },
   437→  ];
   438→  for (const c of cases) {
   439→    await db.evaluationCase.create({ data: { setId: evalSet.id, input: c.input, expected: c.expected, tags: c.tags } });
   440→  }
   441→  created.evaluationCases = cases.length;
   442→
   443→  return { created, skipped: false };
   444→}

   462→async function ensureKnowledge(tenantId: string, opts: {
   463→  sourceType: string; title: string; sourceUri?: string; author?: string;
   464→  trustLevel: string; verificationStatus: string; dataClassification: string;
   465→  claims: Array<{ type: string; claim: string; content: string; scope?: string; validFrom?: Date; evidence?: Array<{ evidenceType: string; content: string; section?: string }> }>;
   466→}) {
   467→  let source = await db.knowledgeSource.findFirst({ where: { tenantId, title: opts.title } });
   468→  if (!source) {
   469→    source = await db.knowledgeSource.create({
   470→      data: {
   471→        tenantId, sourceType: opts.sourceType, title: opts.title, sourceUri: opts.sourceUri,
   472→        author: opts.author, trustLevel: opts.trustLevel, verificationStatus: opts.verificationStatus,
   473→        dataClassification: opts.dataClassification, publishedAt: new Date(),
   474→      },
   475→    });
   476→  }
   477→  const app = await db.application.findFirst({ where: { tenantId } });
   478→  if (!app) return;
   479→  for (const c of opts.claims) {
   480→    const existing = await db.knowledgeItem.findFirst({ where: { tenantId, sourceId: source.id, claim: c.claim } });
   481→    if (existing) continue;
   482→    const k = await db.knowledgeItem.create({
   483→      data: {
   484→        tenantId, applicationId: app.id, sourceId: source.id, type: c.type,
   485→        scope: c.scope ?? "APPLICATION", claim: c.claim, content: c.content,
   486→        contentVector: serializeVector(buildTermVector(c.claim + " " + c.content)),
   487→        status: "ACTIVE", confidence: 0.85, validFrom: c.validFrom, refreshSchedule: "manual",
   488→        lastRefreshedAt: new Date(),
   489→      },
   490→    });
   491→    if (c.evidence) {
   492→      for (const e of c.evidence) {
   493→        await db.knowledgeEvidence.create({
   494→          data: { tenantId, knowledgeItemId: k.id, sourceId: source.id, evidenceType: e.evidenceType, content: e.content, section: e.section },
   495→        });
   496→      }
   497→    }
   498→  }
   499→}
