   301→   301→function FeedbackButtons({ requestId, answer, question }: { requestId: string; answer: string; question?: string }) {
   302→   302→  const [given, setGiven] = React.useState<"thumbs_up" | "thumbs_down" | null>(null);
   303→   303→  const [loading, setLoading] = React.useState(false);
   304→   304→
   305→   305→  async function sendFeedback(signal: "thumbs_up" | "thumbs_down") {
   306→   306→    if (given || loading) return;
   307→   307→    setLoading(true);
   308→   308→    setGiven(signal);
   309→   309→    try {
   310→   310→      await apiPost("/api/brain/feedback", {
   311→   311→        requestId, signal, answer: answer.slice(0, 500), question: question?.slice(0, 500),
   312→   312→      });
   313→   313→    } catch {
   314→   314→      // revert on failure
   315→   315→      setGiven(null);
   316→   316→    } finally {
   317→   317→      setLoading(false);
   318→   318→    }
   319→   319→  }
   320→   320→
   321→   321→  return (
   322→   322→    <div className="flex items-center gap-1">
   323→   323→      <button
   324→   324→        onClick={() => sendFeedback("thumbs_up")}
   325→   325→        disabled={!!given || loading}
   326→   326→        className={cn(
   327→   327→          "flex h-6 w-6 items-center justify-center rounded-md border text-[10px] transition-colors",
   328→   328→          given === "thumbs_up"
   329→   329→            ? "border-emerald-500/40 bg-emerald-500/15 text-emerald-600"
   330→   330→            : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   331→   331→        )}
   332→   332→        title="Good answer"
   333→   333→      >
   334→   334→        <ThumbsUp className="h-3 w-3" />
   335→   335→      </button>
   336→   336→      <button
   337→   337→        onClick={() => sendFeedback("thumbs_down")}
   338→   338→        disabled={!!given || loading}
   339→   339→        className={cn(
   340→   340→          "flex h-6 w-6 items-center justify-center rounded-md border text-[10px] transition-colors",
   341→   341→          given === "thumbs_down"
   342→   342→            ? "border-rose-500/40 bg-rose-500/15 text-rose-600"
   343→   343→            : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   344→   344→        )}
   345→   345→        title="Bad answer"
   346→   346→      >
   347→   347→        <ThumbsDown className="h-3 w-3" />
   348→   348→      </button>
   349→   349→      {given && (
   350→   350→        <span className="text-[10px] text-muted-foreground">
   351→   351→          {given === "thumbs_up" ? "Thanks — helps the Brain learn" : "Thanks — Brain will investigate"}
   352→   352→        </span>
   353→   353→      )}
   354→   354→    </div>
   355→   355→  );
   356→   356→}

   358→   358→function ResponseChips({ state }: { state: BrainStreamState }) {
   359→   359→  const chips: React.ReactNode[] = [];
   360→   360→  if (state.model) {
   361→   361→    chips.push(
   362→   362→      <Chip key="model" icon={<Cpu className="h-3 w-3" />} label={`${state.model.model}${state.model.fallbackUsed ? " (fallback)" : ""}`} tone={state.model.fallbackUsed ? "warn" : "ok"} />
   363→   363→    );
   364→   364→  }
   365→   365→  if (state.research) {
   366→   366→    chips.push(<Chip key="research" icon={<Globe className="h-3 w-3" />} label={`web: ${state.research.ingestedCount} learned`} tone="info" />);
   367→   367→  }
   368→   368→  if (state.verification) {
   369→   369→    chips.push(<Chip key="ver" icon={<ShieldCheck className="h-3 w-3" />} label={state.verification.status} tone={verificationTone(state.verification.status)} />);
   370→   370→  }
   371→   371→  if (state.tools.length > 0) {
   372→   372→    chips.push(<Chip key="tools" icon={<Wrench className="h-3 w-3" />} label={`${state.tools.length} tool`} tone="ok" />);
   373→   373→  }
   374→   374→  if (state.evidence.length > 0) {
   375→   375→    chips.push(<Chip key="ev" icon={<FileText className="h-3 w-3" />} label={`${state.evidence.length} evidence`} tone="ok" />);
   376→   376→  }
   377→   377→  if (state.cost) {
   378→   378→    chips.push(<Chip key="cost" icon={<Zap className="h-3 w-3" />} label={`${state.cost.latencyMs}ms`} tone="muted" />);
   379→   379→    if (state.cost.costUsd > 0) chips.push(<Chip key="cost2" icon={<DollarSign className="h-3 w-3" />} label={`$${state.cost.costUsd.toFixed(4)}`} tone="muted" />);
   380→   380→  }
   381→   381→  if (chips.length === 0) return null;
   382→   382→  return <div className="flex flex-wrap items-center gap-1">{chips}</div>;

   530→   530→function ToolCard({ tool }: { tool: ToolResult }) {
   531→   531→  const [approving, setApproving] = React.useState(false);
   532→   532→  const [current, setCurrent] = React.useState(tool);
   533→   533→  React.useEffect(() => { setCurrent(tool); }, [tool]);
   534→   534→  async function approve() {
   535→   535→    setApproving(true);
   536→   536→    try {
   537→   537→      const r = await apiPost<{ result: ToolResult }>("/api/brain/tools/approve", { executionId: current.executionId, approver: "alice@acme.test" });
   538→   538→      setCurrent(r.result);
   539→   539→      toast.success(`Approved ${current.toolId} → ${r.result.state}`);
   540→   540→    } catch (e: any) {
   541→   541→      toast.error(e?.message ?? "approval failed");
   542→   542→    } finally {
   543→   543→      setApproving(false);
   544→   544→    }
   545→   545→  }
   546→   546→  return (
   547→   547→    <div className="rounded-lg border bg-card p-2.5">
   548→   548→      <div className="mb-1 flex items-center justify-between gap-2">
   549→   549→        <div className="flex items-center gap-1.5">

   551→   551→          <span className="text-xs font-medium">{current.toolId}</span>
   552→   552→        </div>
   553→   553→        <Badge variant="outline" className={cn("text-[9px]", actionStateClass(current.state))}>{current.state}</Badge>
   554→   554→      </div>
   555→   555→      {current.output && (
   556→   556→        <pre className="mt-1 max-h-32 overflow-auto rounded bg-muted/40 p-2 text-[10px] font-mono leading-tight">{JSON.stringify(current.output, null, 2)}</pre>
   557→   557→      )}
   558→   558→      {current.error && <p className="mt-1 text-[10px] text-rose-600">{current.error}</p>}
   559→   559→      {current.requiresApproval && !current.approved && current.state === "AUTHORIZED" && (
