    47→export function BrainWidget() {
    48→  const [mode, setMode] = React.useState<BrainMode>("auto");
    49→  const [platformSlug, setPlatformSlug] = React.useState<string>("mashahd");
    50→  const [platforms, setPlatforms] = React.useState<Array<{ slug: string; displayName: string; domain: string; status: string }>>([]);
    51→  const [input, setInput] = React.useState("");
    52→  const [messages, setMessages] = React.useState<ChatMessage[]>([]);
    53→  const [activeMessageId, setActiveMessageId] = React.useState<string | null>(null);
    54→  const [streaming, setStreaming] = React.useState(false);
    55→  const abortRef = React.useRef<AbortController | null>(null);
    56→  const scrollRef = React.useRef<HTMLDivElement>(null);
    57→
    58→  React.useEffect(() => {
    59→    apiGet<{ platforms: Array<{ slug: string; displayName: string; domain: string; status: string }> }>("/api/brain/platforms")
    60→      .then((d) => setPlatforms(d.platforms.filter((p) => p.status === "ACTIVE")))
    61→      .catch(() => {});
    62→  }, []);
    63→
    64→  React.useEffect(() => {
    65→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" });
    66→  }, [messages, activeMessageId]);
    67→
    68→  async function send(text?: string) {
    69→    const content = (text ?? input).trim();
    70→    if (!content || streaming) return;
    71→    setInput("");
    72→    const userMsg: ChatMessage = { id: crypto.randomUUID(), role: "user", content, createdAt: Date.now() };
    73→    const assistantMsg: ChatMessage = { id: crypto.randomUUID(), role: "assistant", content: "", state: initialStreamState(), question: content, createdAt: Date.now() };
    74→    setMessages((m) => [...m, userMsg, assistantMsg]);
    75→    setActiveMessageId(assistantMsg.id);
    76→    setStreaming(true);
    77→    abortRef.current = new AbortController();
    78→    try {
    79→      await streamBrainResponse(
    80→        { input: { text: content }, mode, platformSlug },
    81→        (state) => {
    82→          setMessages((prev) => prev.map((m) => (m.id === assistantMsg.id ? { ...m, content: state.tokens || (state.error ? `⚠️ ${state.error.message}` : ""), state: { ...state } } : m)));
    83→        },
    84→        abortRef.current.signal,
    85→      );
    86→    } catch (err: any) {
    87→      setMessages((prev) => prev.map((m) => (m.id === assistantMsg.id ? { ...m, content: `⚠️ Stream failed: ${err?.message ?? err}` } : m)));
    88→    } finally {
    89→      setStreaming(false);
    90→      // Keep activeMessageId pointing at the last assistant message so the
    91→      // trace / evidence / tools / memory panels stay populated.
    92→    }
    93→  }
    94→
    95→  function stop() {
    96→    abortRef.current?.abort();
    97→    setStreaming(false);
    98→  }
    99→
   100→  return (
   101→    <div className="grid h-full grid-cols-1 gap-4 lg:grid-cols-[minmax(0,1fr)_minmax(0,1.2fr)] xl:grid-cols-[minmax(0,1fr)_minmax(0,1.4fr)_minmax(0,1fr)]">
   102→      {/* Left: chat */}
   103→      <Card className="flex h-[calc(100vh-9.5rem)] flex-col overflow-hidden">
   104→        <CardHeader className="border-b pb-4">
   105→          <div className="flex items-center justify-between gap-2">
   106→            <div className="flex items-center gap-2">
   107→              <div className="relative flex h-9 w-9 items-center justify-center rounded-lg cirkle-gradient text-white shadow-sm cirkle-glow">
   108→                <CirkleMark size={28} className="drop-shadow-[0_0_4px_rgba(194,160,96,0.4)]" />
   109→                <span className="absolute -bottom-0.5 -right-0.5 flex h-3 w-3 items-center justify-center rounded-full bg-[color:var(--color-cirkle-cyan)] ring-2 ring-background">
   110→                  <span className="h-1.5 w-1.5 cirkle-pulse rounded-full bg-[color:var(--color-cirkle-cyan-deep)]" />
   111→                </span>
   112→              </div>
   113→              <div>
   114→                <CardTitle className="text-base leading-tight">Cirkle Brain AI</CardTitle>
   115→                <CardDescription className="text-[11px] leading-tight">Cognitive operating layer · model-independent</CardDescription>
   116→              </div>
   117→            </div>
   118→            <div className="flex items-center gap-1.5">
   119→              <PlatformSelector value={platformSlug} onChange={setPlatformSlug} platforms={platforms} disabled={streaming} />
   120→              <ModeSelector value={mode} onChange={setMode} disabled={streaming} />
   121→            </div>
   122→          </div>
   123→        </CardHeader>
   124→        <CardContent className="flex-1 overflow-hidden p-0">
   125→          <div ref={scrollRef} className="h-full overflow-y-auto p-4">
   126→            {messages.length === 0 ? (
   127→              <EmptyState onPick={(p) => send(p)} />
   128→            ) : (
   129→              <div className="space-y-4">
   130→                {messages.map((m) => (
   131→                  <ChatBubble key={m.id} message={m} active={m.id === activeMessageId} />
   132→                ))}
   133→              </div>
   134→            )}
   135→          </div>
   136→        </CardContent>
   137→        <div className="border-t p-3">
   138→          <div className="flex items-end gap-2">
   139→            <Textarea
   140→              value={input}
   141→              onChange={(e) => setInput(e.target.value)}
   142→              onKeyDown={(e) => { if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); send(); } }}
   143→              placeholder="Ask the Brain…  (Enter to send, Shift+Enter for newline)"
   144→              className="min-h-[44px] max-h-40 resize-none"
   145→              disabled={streaming}

   153→                <Send className="h-4 w-4" />

   155→            )}
   156→          </div>
   157→        </div>
   158→      </Card>
   159→
   160→      {/* Middle: cognitive trace */}
   161→      <CognitiveTrace messages={messages} activeId={activeMessageId} />
   162→
   163→      {/* Right: admin console (capabilities/health/metrics/audit/memory/knowledge/candidates) */}
   164→      <Card className="hidden h-[calc(100vh-9.5rem)] flex-col overflow-hidden xl:flex">
   165→        <CardHeader className="border-b pb-3">
   166→          <CardTitle className="flex items-center gap-2 text-sm"><Network className="h-4 w-4" /> Admin Console</CardTitle>
   167→        </CardHeader>
   168→        <CardContent className="flex-1 overflow-hidden p-0">
   169→          <AdminConsole />
   170→        </CardContent>

   282→            <div className="flex items-center gap-2 rounded-2xl rounded-tl-sm border bg-card px-3 py-2 text-xs text-muted-foreground">

   285→            </div>
   286→          ) : null}
   287→          {state && !streaming && <ResponseChips state={state} />}
   288→          {state && !streaming && state.response && (
   289→            <FeedbackButtons
   290→              requestId={state.response.requestId}
   291→              answer={message.content}
   292→              question={message.question}
   293→            />
   294→          )}
   295→        </div>
   296→      </div>
   297→    </div>
   298→  );

   301→function FeedbackButtons({ requestId, answer, question }: { requestId: string; answer: string; question?: string }) {
   302→  const [given, setGiven] = React.useState<"thumbs_up" | "thumbs_down" | null>(null);
   303→  const [loading, setLoading] = React.useState(false);
   304→
   305→  async function sendFeedback(signal: "thumbs_up" | "thumbs_down") {
   306→    if (given || loading) return;
   307→    setLoading(true);
   308→    setGiven(signal);
   309→    try {
   310→      await apiPost("/api/brain/feedback", {
   311→        requestId, signal, answer: answer.slice(0, 500), question: question?.slice(0, 500),
   312→      });
   313→    } catch {
   314→      // revert on failure
   315→      setGiven(null);
   316→    } finally {
   317→      setLoading(false);
   318→    }
   319→  }
   320→
   321→  return (
   322→    <div className="flex items-center gap-1">
   323→      <button
   324→        onClick={() => sendFeedback("thumbs_up")}
   325→        disabled={!!given || loading}
   326→        className={cn(
   327→          "flex h-6 w-6 items-center justify-center rounded-md border text-[10px] transition-colors",
   328→          given === "thumbs_up"
   329→            ? "border-emerald-500/40 bg-emerald-500/15 text-emerald-600"
   330→            : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   331→        )}
   332→        title="Good answer"
   333→      >
   334→        <ThumbsUp className="h-3 w-3" />
   335→      </button>
   336→      <button
   337→        onClick={() => sendFeedback("thumbs_down")}
   338→        disabled={!!given || loading}
   339→        className={cn(
   340→          "flex h-6 w-6 items-center justify-center rounded-md border text-[10px] transition-colors",
   341→          given === "thumbs_down"
   342→            ? "border-rose-500/40 bg-rose-500/15 text-rose-600"
   343→            : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   344→        )}
   345→        title="Bad answer"
   346→      >
   347→        <ThumbsDown className="h-3 w-3" />
   348→      </button>
   349→      {given && (
   350→        <span className="text-[10px] text-muted-foreground">
   351→          {given === "thumbs_up" ? "Thanks — helps the Brain learn" : "Thanks — Brain will investigate"}
   352→        </span>
   353→      )}
   354→    </div>
   355→  );
   356→}

   358→function ResponseChips({ state }: { state: BrainStreamState }) {
   359→  const chips: React.ReactNode[] = [];
   360→  if (state.model) {
   361→    chips.push(
   362→      <Chip key="model" icon={<Cpu className="h-3 w-3" />} label={`${state.model.model}${state.model.fallbackUsed ? " (fallback)" : ""}`} tone={state.model.fallbackUsed ? "warn" : "ok"} />
   363→    );
   364→  }
   365→  if (state.research) {
   366→    chips.push(<Chip key="research" icon={<Globe className="h-3 w-3" />} label={`web: ${state.research.ingestedCount} learned`} tone="info" />);
   367→  }
   368→  if (state.verification) {
   369→    chips.push(<Chip key="ver" icon={<ShieldCheck className="h-3 w-3" />} label={state.verification.status} tone={verificationTone(state.verification.status)} />);
   370→  }
   371→  if (state.tools.length > 0) {
   372→    chips.push(<Chip key="tools" icon={<Wrench className="h-3 w-3" />} label={`${state.tools.length} tool`} tone="ok" />);
   373→  }
   374→  if (state.evidence.length > 0) {
   375→    chips.push(<Chip key="ev" icon={<FileText className="h-3 w-3" />} label={`${state.evidence.length} evidence`} tone="ok" />);
   376→  }
   377→  if (state.cost) {
   378→    chips.push(<Chip key="cost" icon={<Zap className="h-3 w-3" />} label={`${state.cost.latencyMs}ms`} tone="muted" />);
   379→    if (state.cost.costUsd > 0) chips.push(<Chip key="cost2" icon={<DollarSign className="h-3 w-3" />} label={`$${state.cost.costUsd.toFixed(4)}`} tone="muted" />);
   380→  }
   381→  if (chips.length === 0) return null;
   382→  return <div className="flex flex-wrap items-center gap-1">{chips}</div>;
   383→}

   385→function Chip({ icon, label, tone }: { icon: React.ReactNode; label: string; tone: "ok" | "warn" | "err" | "muted" | "info" }) {
   386→  const cls = {
   387→    ok: "border-[color:var(--color-cirkle-cyan)]/30 bg-[color:var(--color-cirkle-cyan)]/10 text-[color:var(--color-cirkle-cyan-deep)] dark:text-[color:var(--color-cirkle-glow)]",
   388→    info: "border-[color:var(--color-cirkle-cyan)]/40 bg-[color:var(--color-cirkle-cyan)]/15 text-[color:var(--color-cirkle-cyan-deep)] dark:text-[color:var(--color-cirkle-glow)]",
   389→    warn: "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-300",
   390→    err: "border-rose-500/30 bg-rose-500/10 text-rose-700 dark:text-rose-300",
   391→    muted: "border-border bg-muted text-muted-foreground",
   392→  }[tone];
   393→  return (
   394→    <span className={cn("inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-[10px] font-medium", cls)}>
   395→      {icon}
   396→      {label}
   397→    </span>
   398→  );
   399→}

   401→function verificationTone(s: EvidenceStatus): "ok" | "warn" | "err" | "muted" {
   402→  if (s === "VERIFIED") return "ok";
   403→  if (s === "SUPPORTED") return "ok";
   404→  if (s === "INFERRED") return "warn";
   405→  if (s === "UNCERTAIN") return "warn";
   406→  if (s === "CONFLICTED") return "warn";
   407→  if (s === "UNSUPPORTED" || s === "UNKNOWN") return "err";
   408→  return "muted";
   409→}

   416→  const active = messages.find((m) => m.id === activeId && m.role === "assistant");
   417→  const state = active?.state;

   420→      <CardHeader className="border-b pb-3">

   422→        <CardDescription className="text-[11px]">Real-time execution path — no private hidden reasoning exposed</CardDescription>
   423→      </CardHeader>
   424→      <CardContent className="flex-1 overflow-hidden p-0">
   425→        <Tabs defaultValue="trace" className="flex h-full flex-col">
   426→          <TabsList className="mx-3 mt-2 grid grid-cols-5">
   427→            <TabsTrigger value="trace" className="text-xs">Trace</TabsTrigger>
   428→            <TabsTrigger value="evidence" className="text-xs">Evidence</TabsTrigger>
   429→            <TabsTrigger value="research" className="text-xs">Research</TabsTrigger>
   430→            <TabsTrigger value="tools" className="text-xs">Tools</TabsTrigger>
   431→            <TabsTrigger value="memory" className="text-xs">Memory</TabsTrigger>
   432→          </TabsList>
   433→          <ScrollArea className="flex-1">
   434→            <div className="p-3">
   435→              {!state ? (

   530→function ToolCard({ tool }: { tool: ToolResult }) {
   531→  const [approving, setApproving] = React.useState(false);
   532→  const [current, setCurrent] = React.useState(tool);
   533→  React.useEffect(() => { setCurrent(tool); }, [tool]);
   534→  async function approve() {
   535→    setApproving(true);
   536→    try {
   537→      const r = await apiPost<{ result: ToolResult }>("/api/brain/tools/approve", { executionId: current.executionId, approver: "alice@acme.test" });
   538→      setCurrent(r.result);
   539→      toast.success(`Approved ${current.toolId} → ${r.result.state}`);
   540→    } catch (e: any) {
   541→      toast.error(e?.message ?? "approval failed");
   542→    } finally {
   543→      setApproving(false);
   544→    }
   545→  }
   546→  return (
   547→    <div className="rounded-lg border bg-card p-2.5">
   548→      <div className="mb-1 flex items-center justify-between gap-2">
   549→        <div className="flex items-center gap-1.5">
   550→          <Wrench className="h-3 w-3 text-[color:var(--color-cirkle-cyan)]" />
   551→          <span className="text-xs font-medium">{current.toolId}</span>
   552→        </div>
   553→        <Badge variant="outline" className={cn("text-[9px]", actionStateClass(current.state))}>{current.state}</Badge>
   554→      </div>
   555→      {current.output && (
   556→        <pre className="mt-1 max-h-32 overflow-auto rounded bg-muted/40 p-2 text-[10px] font-mono leading-tight">{JSON.stringify(current.output, null, 2)}</pre>
   557→      )}
   558→      {current.error && <p className="mt-1 text-[10px] text-rose-600">{current.error}</p>}
   559→      {current.requiresApproval && !current.approved && current.state === "AUTHORIZED" && (
   560→        <Button size="sm" className="mt-2 h-7 w-full text-xs" onClick={approve} disabled={approving}>
   561→          {approving ? <Loader2 className="mr-1 h-3 w-3 animate-spin" /> : <ShieldCheck className="mr-1 h-3 w-3" />}
   562→          Approve (human authorization)
   563→        </Button>
   564→      )}
   565→      <div className="mt-1 flex items-center justify-between text-[10px] text-muted-foreground">
   566→        <span>{current.durationMs}ms</span>
   567→        <span>governed execution</span>
   568→      </div>
   569→    </div>
   570→  );
   571→}

   626→function verificationToneClass(s: EvidenceStatus) {
   627→  if (s === "VERIFIED" || s === "SUPPORTED") return "border-[color:var(--color-cirkle-cyan)]/30 bg-[color:var(--color-cirkle-cyan)]/10 text-[color:var(--color-cirkle-cyan-deep)] dark:text-[color:var(--color-cirkle-glow)]";
   628→  if (s === "INFERRED" || s === "UNCERTAIN" || s === "CONFLICTED") return "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-300";
   629→  return "border-rose-500/30 bg-rose-500/10 text-rose-700 dark:text-rose-300";
   630→}

   632→function actionStateClass(s: string) {
   633→  if (s === "VERIFIED" || s === "EXECUTED") return "border-[color:var(--color-cirkle-cyan)]/30 bg-[color:var(--color-cirkle-cyan)]/10 text-[color:var(--color-cirkle-cyan-deep)] dark:text-[color:var(--color-cirkle-glow)]";
   634→  if (s === "AUTHORIZED") return "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-300";
   635→  if (s === "FAILED" || s === "TIMED_OUT" || s === "REJECTED") return "border-rose-500/30 bg-rose-500/10 text-rose-700 dark:text-rose-300";
   636→  return "border-border bg-muted text-muted-foreground";
   637→}
