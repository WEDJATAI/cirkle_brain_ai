   355→   355→function FeedbackButtons({ requestId, answer, question }: { requestId: string; answer: string; question?: string }) {
   356→   356→  const [given, setGiven] = React.useState<"thumbs_up" | "thumbs_down" | null>(null);
   357→   357→  const [loading, setLoading] = React.useState(false);
   358→   358→
   359→   359→  async function sendFeedback(signal: "thumbs_up" | "thumbs_down") {
   360→   360→    if (given || loading) return;
   361→   361→    setLoading(true);
   362→   362→    setGiven(signal);
   363→   363→    try {
   364→   364→      await apiPost("/api/brain/feedback", {
   365→   365→        requestId, signal, answer: answer.slice(0, 500), question: question?.slice(0, 500),
   366→   366→      });
   367→   367→    } catch {
   368→   368→      // revert on failure
   369→   369→      setGiven(null);
   370→   370→    } finally {
   371→   371→      setLoading(false);
   372→   372→    }
   373→   373→  }
   374→   374→
   375→   375→  return (
   376→   376→    <div className="flex items-center gap-1">
   377→   377→      <button
   378→   378→        onClick={() => sendFeedback("thumbs_up")}
   379→   379→        disabled={!!given || loading}
   380→   380→        className={cn(
   381→   381→          "flex h-6 w-6 items-center justify-center rounded-md border text-[10px] transition-colors",
   382→   382→          given === "thumbs_up"
   383→   383→            ? "border-emerald-500/40 bg-emerald-500/15 text-emerald-600"
   384→   384→            : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   385→   385→        )}
   386→   386→        title="Good answer"
   387→   387→      >
